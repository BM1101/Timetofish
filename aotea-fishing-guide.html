/** * Aotea tides proxy — Cloudflare Worker * * Sits between the fishing-guide page and the NIWA Tide API so the API key * never ships to the browser. Only answers for coordinates on/around Aotea * Great Barrier Island, and caches each rounded coordinate + start date for * 12 hours, so many visitors cost about one NIWA call per spot per day. * * Setup: * 1. Create an API key at https://developer.niwa.co.nz with Tide API access. * 2. Cloudflare dashboard → Workers → create → paste this file. * 3. Settings → Variables → add secret NIWA_API_KEY = * Optionally add ALLOWED_ORIGIN = https://your-site.example (default: *) * 4. Deploy, copy the workers.dev URL into TIDE_API in the HTML page. * * Request: GET /?lat=-36.32&long=175.48&start=2026-09-06&days=12 * Response: NIWA's JSON passed through ({ metadata, values:[{time,value}] }) */ const TIDE_API = 'https://misty-heart-b3d1.benmason04-d24.workers.dev'; const BOX = { latMin: -36.45, latMax: -36.0, lonMin: 175.2, lonMax: 175.7 }; const CACHE_SECONDS = 12 * 3600; export default { async fetch(request, env, ctx) { const origin = env.ALLOWED_ORIGIN || '*'; const cors = { 'Access-Control-Allow-Origin': origin, 'Access-Control-Allow-Methods': 'GET, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', }; if (request.method === 'OPTIONS') return new Response(null, { headers: cors }); if (request.method !== 'GET') return json({ error: 'GET only' }, 405, cors); const u = new URL(request.url); const lat = Number(u.searchParams.get('lat')); const long = Number(u.searchParams.get('long')); const start = u.searchParams.get('start') || ''; const days = Math.min(Math.max(parseInt(u.searchParams.get('days') || '7', 10), 1), 31); if (!Number.isFinite(lat) || !Number.isFinite(long)) return json({ error: 'lat and long are required' }, 400, cors); if (lat < BOX.latMin || lat > BOX.latMax || long < BOX.lonMin || long > BOX.lonMax) return json({ error: 'This service only covers Aotea Great Barrier Island' }, 403, cors); if (start && !/^\d{4}-\d{2}-\d{2}$/.test(start)) return json({ error: 'start must be YYYY-MM-DD' }, 400, cors); if (!env.NIWA_API_KEY) return json({ error: 'Worker is missing NIWA_API_KEY' }, 500, cors); // Round coordinates so nearby requests share a cache entry. const rl = lat.toFixed(3), rg = long.toFixed(3); const cacheKey = new Request(`https://aotea-tides.cache/${rl}/${rg}/${start || 'today'}/${days}`); const cache = caches.default; const hit = await cache.match(cacheKey); if (hit) return withHeaders(hit, cors, 'HIT'); const q = new URLSearchParams({ lat: rl, long: rg, numberOfDays: String(days), datum: 'LAT' }); if (start) q.set('startDate', start); let upstream; try { upstream = await fetch(`${NIWA}?${q}`, { headers: { Accept: 'application/json', 'x-apikey': env.NIWA_API_KEY } }); } catch (e) { return json({ error: 'Could not reach NIWA' }, 502, cors); } if (!upstream.ok) return json({ error: `NIWA responded ${upstream.status}` }, 502, cors); const body = await upstream.text(); const res = new Response(body, { status: 200, headers: { 'Content-Type': 'application/json', 'Cache-Control': `public, max-age=${CACHE_SECONDS}` }, }); ctx.waitUntil(cache.put(cacheKey, res.clone())); return withHeaders(res, cors, 'MISS'); }, }; function json(obj, status, cors) { return new Response(JSON.stringify(obj), { status, headers: { 'Content-Type': 'application/json', ...cors } }); } function withHeaders(res, cors, cacheState) { const r = new Response(res.body, res); for (const [k, v] of Object.entries(cors)) r.headers.set(k, v); r.headers.set('X-Cache', cacheState); return r; }
